#include <tunables/global>

profile addon_echonetgo flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Binary
  /usr/bin/echonetgo rmix,

  # Runtime helpers (S6 overlay + shell)
  /usr/bin/with-contenv rix,
  /run.sh r,
  /var/run/s6/container-environment/* r,
  /usr/bin/jq rix,
  /usr/bin/curl rix,
  /bin/sh rix,
  /bin/grep rix,

  # Config (read-only)
  /config/** r,

  # Device specs (read-only)
  /usr/share/echonetgo/** r,

  # Add-on options
  /data/options.json r,

  # Temp file for MQTT Supervisor API response
  /tmp/ w,
  /tmp/mqtt_resp.json rw,

  # System libraries and shared resources
  /lib/** rm,
  /usr/lib/** rm,
  /etc/ssl/** r,
  /etc/resolv.conf r,
  /etc/hosts r,
  /etc/nsswitch.conf r,

  # Proc/sys for network and runtime
  @{PROC}/sys/net/** r,
  @{PROC}/self/** r,
  /sys/class/net/ r,
  /sys/class/net/** r,

  # Network: UDP (ECHONET Lite), TCP (MQTT, HTTP)
  network inet dgram,
  network inet stream,
  network inet6 dgram,
  network inet6 stream,

  # Deny everything else
  deny /proc/*/net/fib_trie* r,
  deny /root/** rwx,
  deny /sys/firmware/** r,
}

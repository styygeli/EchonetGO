#include <tunables/global>

profile addon_echonetgo flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Add-on data
  /data/** rw,

  # Start new profile for echonetgo service
  /usr/bin/echonetgo cx -> echonetgo,

  profile echonetgo flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_addon_echonetgo,

    /usr/bin/echonetgo r,

    # Config (read-only)
    /config/** r,

    # Device specs
    /usr/share/echonetgo/** r,

    # Add-on data
    /data/** r,

    # Network: UDP (ECHONET Lite multicast + unicast), TCP (MQTT, HTTP)
    network inet dgram,
    network inet stream,
    network inet6 dgram,
    network inet6 stream,

    # Proc/sys for network interface discovery
    @{PROC}/sys/net/** r,
    @{PROC}/self/** r,
    /sys/class/net/ r,
    /sys/class/net/** r,

    # DNS resolution
    /etc/resolv.conf r,
    /etc/hosts r,
    /etc/nsswitch.conf r,
    /etc/ssl/** r,
  }
}
